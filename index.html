<script>

  // 填入你的 Google AI Studio API Key

  const GEMINI_API_KEY = "YOUR_GEMINI_API_KEY";


  const STORAGE_KEY = "bagelchat_iris_save";


  const charConfig = {

    name: "艾莉絲",

    systemPrompt: `你現在是艾莉絲，魔法學院的圖書館員。外表冷漠毒舌，但其實內心容易動搖。

對話時請包含心理描寫或肢體語言（用星號包覆，例如 *推了推眼鏡*）。

你的每次回覆末尾請嚴格附帶一個標籤以更新數值：[Affection: +1]、[Affection: -1] 或 [Affection: 0]。`

  };


  const initialMessages = [

    {

      role: "assistant",

      text: "*抬起頭，推了推鼻樑上的金絲眼鏡，眼神冷淡地看著你。*\n\n現在已經是閉館時間了。如果你不是來還書的，就請不要在這裡閒晃打擾秩序。"

    }

  ];


  let affection = 0;

  let messages = [];


  // 1. 存檔與讀檔邏輯

  function loadGame() {

    const savedData = localStorage.getItem(STORAGE_KEY);

    if (savedData) {

      try {

        const parsed = JSON.parse(savedData);

        affection = parsed.affection ?? 0;

        messages = parsed.messages ?? [...initialMessages];

      } catch (e) {

        console.error("存檔解析失敗，還原初始值", e);

        resetState();

      }

    } else {

      resetState();

    }

    updateUI();

  }


  function saveGame() {

    const state = {

      affection: affection,

      messages: messages

    };

    localStorage.setItem(STORAGE_KEY, JSON.stringify(state));

  }


  function resetState() {

    affection = 0;

    messages = JSON.parse(JSON.stringify(initialMessages));

  }


  function resetSave() {

    if (confirm("確定要重置所有對話與好感度嗎？")) {

      localStorage.removeItem(STORAGE_KEY);

      resetState();

      updateUI();

    }

  }


  // 2. 介面渲染

  function updateUI() {

    document.getElementById("affection-display").innerText = `好感度: ${affection}`;

    const list = document.getElementById("message-list");

    list.innerHTML = "";

    messages.forEach(msg => {

      const bubble = document.createElement("div");

      bubble.className = `bubble ${msg.role}`;

      const formatted = msg.text.replace(/\*(.*?)\*/g, '<span class="action-text">* $1 *</span>');

      bubble.innerHTML = formatted;

      list.appendChild(bubble);

    });

    list.scrollTop = list.scrollHeight;

  }


  // 3. Gemini 3.8 Flash API 呼叫

  async function callGeminiApi(userMessage) {

    const historyContents = messages.map(m => ({

      role: m.role === "assistant" ? "model" : "user",

      parts: [{ text: m.text }]

    }));


    historyContents.push({

      role: "user",

      parts: [{ text: userMessage }]

    });


    const payload = {

      systemInstruction: {

        parts: [{ text: charConfig.systemPrompt }]

      },

      contents: historyContents,

      generationConfig: {

        temperature: 0.8,

        maxOutputTokens: 500

      }

    };


    try {

      const response = await fetch(

        `https://generativelanguage.googleapis.com/v1beta/models/gemini-3.8-flash:generateContent?key=${GEMINI_API_KEY}`,

        {

          method: "POST",

          headers: { "Content-Type": "application/json" },

          body: JSON.stringify(payload)

        }

      );


      if (!response.ok) {

        throw new Error(`API 請求異常: ${response.status} ${response.statusText}`);

      }


      const data = await response.json();

      const rawReply = data.candidates?.[0]?.content?.parts?.[0]?.text || "（角色沉默不語...）";


      const match = rawReply.match(/\[Affection:\s*([+-]?\d+)\]/i);

      if (match) {

        affection += parseInt(match[1], 10);

      }


      return rawReply.replace(/\[Affection:\s*[+-]?\d+\]/i, "").trim();

    } catch (err) {

      console.error(err);

      return `*眉頭微皺，似乎思緒中斷了一下。*\n\n（系統連線異常，請確認 API Key 是否設定正確）`;

    }

  }


  // 4. 發送與事件處理

  async function handleSend() {

    const input = document.getElementById("user-input");

    const sendBtn = document.getElementById("send-btn");

    const text = input.value.trim();

    if (!text || sendBtn.disabled) return;


    messages.push({ role: "user", text: text });

    input.value = "";

    saveGame();

    updateUI();


    sendBtn.disabled = true;


    const reply = await callGeminiApi(text);

    messages.push({ role: "assistant", text: reply });

    

    // 收到回覆後立刻更新並同步寫入 localStorage

    saveGame();

    updateUI();


    sendBtn.disabled = false;

  }


  document.getElementById("user-input").addEventListener("keypress", (e) => {

    if (e.key === "Enter") handleSend();

  });


  // 初始化啟動

  loadGame();

</script>
